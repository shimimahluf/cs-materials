public class Product
{
    private string name;
    private double price;
    private int quantity;

    public Product(string name, double price, int quantity)
    {
        this.name = name;
        this.price = price;
        this.quantity = quantity;
    }
    public string GetName() { return name; }
    public double GetPrice() { return price; }
    public int GetQuantity() { return quantity; }
    public void SetQuantity(int quantity) { this.quantity = quantity; }

    public double TotalValue()
    {
        return price * quantity;
    }

    public bool Sell(int amount)
    {
        if (amount > quantity)
            return false;
        quantity -= amount;
        return true;
    }
}

public static string MostValuable(Product[] products)
{
    int best = 0;
    for (int i = 1; i < products.Length; i++)
        if (products[i].TotalValue() > products[best].TotalValue())
            best = i;
    return products[best].GetName();
}

public Classroom(string className, int capacity)
{
    this.className = className;
    this.students = new Student[capacity];
    this.count = 0;
}

public bool Add(Student st)
{
    if (count == students.Length)
        return false;
    students[count] = st;
    count++;
    return true;
}

public double Average()
{
    double sum = 0;
    for (int i = 0; i < count; i++)
        sum += students[i].GetAverage();
    return sum / count;
}

public int CountAbove(double g)
{
    int c = 0;
    for (int i = 0; i < count; i++)
        if (students[i].GetAverage() > g)
            c++;
    return c;
}

public static string BestClass(Classroom[] classes)
{
    Classroom best = classes[0];
    for (int i = 1; i < classes.Length; i++)
        if (classes[i].Average() > best.Average())
            best = classes[i];
    return best.GetClassName();
}

public Train(int number, string destination, int depHour, int depMin, int arrHour, int arrMin)
{
    this.number = number;
    this.destination = destination;
    this.departure = new Time(depHour, depMin);   // the inner objects are created here
    this.arrival = new Time(arrHour, arrMin);
}

public int Duration()
{
    return arrival.ToMinutes() - departure.ToMinutes();
}

private static void AddMinutes(Time t, int minutes)
{
    int total = t.ToMinutes() + minutes;
    t.SetHour(total / 60);
    t.SetMinute(total % 60);
}

public void Delay(int minutes)
{
    AddMinutes(departure, minutes);
    AddMinutes(arrival, minutes);
}

// e2: deep copy
public Train(Train other)
{
    this.number = other.number;
    this.destination = other.destination;
    this.departure = new Time(other.departure.GetHour(), other.departure.GetMinute());
    this.arrival = new Time(other.arrival.GetHour(), other.arrival.GetMinute());
}

public int CheckIn(Guest g, int maxPrice)
{
    for (int i = 0; i < rooms.Length; i++)
    {
        if (rooms[i].IsFree() && rooms[i].GetPrice() <= maxPrice)
        {
            rooms[i].SetGuest(g);
            return rooms[i].GetNumber();
        }
    }

public int CheckOut(string name)
{
    for (int i = 0; i < rooms.Length; i++)
    {
        Guest g = rooms[i].GetGuest();
        if (g != null && g.GetName() == name)
        {
            int pay = g.GetNights() * rooms[i].GetPrice();
            rooms[i].SetGuest(null);
            return pay;
        }
    }
    return -1;
}

public int FreeRooms()
{
    int c = 0;
    for (int i = 0; i < rooms.Length; i++)
        if (rooms[i].IsFree())
            c++;
    return c;
}

public Guest LongestStay()
{
    Guest best = null;
    for (int i = 0; i < rooms.Length; i++)
    {
        Guest g = rooms[i].GetGuest();
        if (g != null && (best == null || g.GetNights() > best.GetNights()))
            best = g;
    }
    return best;
}

public static int TotalTime(Song[] songs, string artist)
{
    int sum = 0;
    for (int i = 0; i < songs.Length; i++)
        if (songs[i].GetArtist() == artist)
            sum += songs[i].GetSeconds();
    return sum;
}

public static string TopArtist(Song[] songs)
{
    string best = songs[0].GetArtist();
    int max = TotalTime(songs, best);
    for (int i = 1; i < songs.Length; i++)
    {
        int t = TotalTime(songs, songs[i].GetArtist());
        if (t > max)
        {
            max = t;
            best = songs[i].GetArtist();
        }
    }
    return best;
}

public static bool HasDuplicates(Song[] songs)
{
    for (int i = 0; i < songs.Length; i++)
        for (int j = i + 1; j < songs.Length; j++)
            if (songs[i].GetTitle() == songs[j].GetTitle() &&
                songs[i].GetArtist() == songs[j].GetArtist())
                return true;
    return false;
}

public static Song[] LongerThan(Song[] songs, int sec)
{
    int count = 0;
    for (int i = 0; i < songs.Length; i++)
        if (songs[i].GetSeconds() > sec)
            count++;
    Song[] result = new Song[count];
    int k = 0;
    for (int i = 0; i < songs.Length; i++)
    {
        if (songs[i].GetSeconds() > sec)
        {
            result[k] = songs[i];
            k++;
        }
    }
    return result;
}
