// in Date
public bool IsBefore(Date other)
{
    if (year != other.year)
        return year < other.year;
    if (month != other.month)
        return month < other.month;
    return day < other.day;
}

public Date AddMonths(int months)
{
    int total = (month - 1) + months;          // months counted from January = 0
    return new Date(day, total % 12 + 1, year + total / 12);
}

public int CountExpired(Date today)
{
    int c = 0;
    for (int i = 0; i < count; i++)
        if (members[i].GetExpiry().IsBefore(today))
            c++;
    return c;
}

public Member[] ExpiredMembers(Date today)
{
    Member[] result = new Member[CountExpired(today)];
    int k = 0;
    for (int i = 0; i < count; i++)
    {
        if (members[i].GetExpiry().IsBefore(today))
        {
            result[k] = members[i];
            k++;
        }
    }
    return result;
}

public bool Renew(string name, int months)
{
    for (int i = 0; i < count; i++)
    {
        if (members[i].GetName() == name)
        {
            members[i].SetExpiry(members[i].GetExpiry().AddMonths(months));
            return true;
        }
    }
    return false;
}

public bool InsertAt(int index, Song s)
{
    if (count == songs.Length || index < 0 || index > count)
        return false;
    for (int i = count; i > index; i--)        // from the end, so nothing is overwritten
        songs[i] = songs[i - 1];
    songs[index] = s;
    count++;
    return true;
}

public Song RemoveAt(int index)
{
    if (index < 0 || index >= count)
        return null;
    Song removed = songs[index];
    for (int i = index; i < count - 1; i++)
        songs[i] = songs[i + 1];
    songs[count - 1] = null;
    count--;
    return removed;
}

public int RemoveArtist(string artist)
{
    int removed = 0;
    int i = 0;
    while (i < count)
    {
        if (songs[i].GetArtist() == artist)
        {
            RemoveAt(i);          // the next song moved into cell i: don't advance
            removed++;
        }
        else
            i++;
    }
    return removed;
}

public void MoveToFront(string title)
{
    for (int i = 0; i < count; i++)
    {
        if (songs[i].GetTitle() == title)
        {
            Song s = RemoveAt(i);
            InsertAt(0, s);
            return;
        }
    }
}

public double Salary(Worker w)
{
    int hours = 0;
    for (int i = 0; i < count; i++)
        if (shifts[i].GetWorker() == w)          // the same object
            hours += shifts[i].GetHours();
    return hours * w.GetWage();
}

public int[] HoursPerDay()
{
    int[] h = new int[7];
    for (int i = 0; i < count; i++)
        h[shifts[i].GetDay() - 1] += shifts[i].GetHours();
    return h;
}

public int BusiestDay()
{
    int[] h = HoursPerDay();
    int best = 0;
    for (int d = 1; d < 7; d++)
        if (h[d] > h[best])                      // > keeps the earlier day on a tie
            best = d;
    return best + 1;
}

public bool HasConflict()
{
    for (int i = 0; i < count; i++)
        for (int j = i + 1; j < count; j++)
            if (shifts[i].GetWorker() == shifts[j].GetWorker() &&
                shifts[i].GetDay() == shifts[j].GetDay())
                return true;
    return false;
}

public Worker TopEarner()
{
    Worker best = shifts[0].GetWorker();
    for (int i = 1; i < count; i++)
        if (Salary(shifts[i].GetWorker()) > Salary(best))
            best = shifts[i].GetWorker();
    return best;
}

public void AddMatch(Match m)
{
    matches[count] = m;
    count++;
    Team w = m.Winner();
    if (w != null)
        w.AddPoints(3);
    else
    {
        m.GetHome().AddPoints(1);
        m.GetAway().AddPoints(1);
    }
}

public int GoalsFor(Team t)
{
    int goals = 0;
    for (int i = 0; i < count; i++)
    {
        if (matches[i].GetHome() == t)
            goals += matches[i].GetHomeGoals();
        else if (matches[i].GetAway() == t)
            goals += matches[i].GetAwayGoals();
    }
    return goals;
}

public Team Leader()
{
    Team best = teams[0];
    for (int i = 1; i < teams.Length; i++)
    {
        Team t = teams[i];
        if (t.GetPoints() > best.GetPoints() ||
            (t.GetPoints() == best.GetPoints() && GoalsFor(t) > GoalsFor(best)))
            best = t;
    }
    return best;
}

public int MatchesOf(Team t)
{
    int c = 0;
    for (int i = 0; i < count; i++)
        if (matches[i].Involves(t))
            c++;
    return c;
}

public Account Find(int id)
{
    for (int i = 0; i < accounts.Length; i++)
        if (accounts[i].GetId() == id)
            return accounts[i];
    return null;
}

public bool Transfer(int fromId, int toId, double amount)
{
    Account from = Find(fromId);
    Account to = Find(toId);
    if (from == null || to == null || from == to)
        return false;
    if (!from.Withdraw(amount))
        return false;
    to.Deposit(amount);
    return true;
}

public string[] Owners()
{
    int distinct = 0;
    for (int i = 0; i < accounts.Length; i++)
        if (IsFirst(i))
            distinct++;
    string[] result = new string[distinct];
    int k = 0;
    for (int i = 0; i < accounts.Length; i++)
    {
        if (IsFirst(i))
        {
            result[k] = accounts[i].GetOwner();
            k++;
        }
    }
    return result;
}

// true if the owner of accounts[i] does not appear in an earlier cell
private bool IsFirst(int i)
{
    for (int j = 0; j < i; j++)
        if (accounts[j].GetOwner() == accounts[i].GetOwner())
            return false;
    return true;
}

public Account[] SortedByBalance()
{
    Account[] sorted = new Account[accounts.Length];
    for (int i = 0; i < accounts.Length; i++)
        sorted[i] = accounts[i];                     // copy the references
    for (int i = 0; i < sorted.Length - 1; i++)      // selection sort, largest first
    {
        int max = i;
        for (int j = i + 1; j < sorted.Length; j++)
            if (sorted[j].GetBalance() > sorted[max].GetBalance())
                max = j;
        Account t = sorted[i];
        sorted[i] = sorted[max];
        sorted[max] = t;
    }
    return sorted;
}
