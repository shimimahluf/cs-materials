Box
public class Box
{
    private int v;
    public Box(int v) { this.v = v; }
    public int GetV() { return v; }
    public void SetV(int v) { this.v = v; }
}

public static Box Bigger(Box a, Box b)
{
    if (a.GetV() >= b.GetV())
        return a;
    return b;
}

public static void Swap(Box[] arr, int i, int j)
{
    Box t = arr[i];
    arr[i] = arr[j];
    arr[j] = t;
}

public static void SwapValues(Box a, Box b)
{
    int t = a.GetV();
    a.SetV(b.GetV());
    b.SetV(t);
}

public static Box[] Copy(Box[] arr)
{
    Box[] c = new Box[arr.Length];
    for (int i = 0; i < arr.Length; i++)
        c[i] = arr[i];
    return c;
}
