ה. O(n + m): אין לולאה בתוך לולאה. הלולאה הראשונה והשחזור של s1 הם עד n סבבים, השחזור של s2 עד m סבבים.

public static void MoveToTop(Stack<int> s, int x)
{
    Stack<int> temp = new Stack<int>();
    int count = 0;
    while (!s.IsEmpty())
    {
        int val = s.Pop();
        if (val == x)
            count++;
        else
            temp.Push(val);
    }
    while (!temp.IsEmpty())        // the order of the others is restored
        s.Push(temp.Pop());
    for (int i = 0; i < count; i++)
        s.Push(x);
}

public static void MoveMinToTop(Stack<int> s)
{
    Stack<int> temp = new Stack<int>();
    int min = s.Top();
    while (!s.IsEmpty())
    {
        int val = s.Pop();
        if (val < min)
            min = val;
        temp.Push(val);
    }
    while (!temp.IsEmpty())
        s.Push(temp.Pop());
    MoveToTop(s, min);
}

ג. O(n): מציאת המינימום היא מעבר אחד והחזרה (O(n)), ואחריה זימון אחד של MoveToTop, שגם הוא O(n). אין לולאה בתוך לולאה.

public static int RemoveBottom(Stack<int> s)
{
    Stack<int> temp = new Stack<int>();
    while (!s.IsEmpty())
        temp.Push(s.Pop());
    int bottom = temp.Pop();        // the bottom is now on top of temp
    while (!temp.IsEmpty())
        s.Push(temp.Pop());
    return bottom;
}

public static bool IsPalindrome(Stack<int> s)
{
    while (!s.IsEmpty())
    {
        int top = s.Pop();
        if (s.IsEmpty())
            return true;            // one element left in the middle
        int bottom = RemoveBottom(s);
        if (top != bottom)
            return false;
    }
    return true;
}

ג. O(n²): הלולאה רצה עד n/2 סבבים, ובכל סבב מזמנים את RemoveBottom, שעוברת על כל מה שנשאר במחסנית (O(n)). סך הכול O(n²).

public static void InsertSorted(Stack<int> s, int x)
{
    Stack<int> temp = new Stack<int>();
    while (!s.IsEmpty() && s.Top() < x)
        temp.Push(s.Pop());
    s.Push(x);
    while (!temp.IsEmpty())
        s.Push(temp.Pop());
}

public static Stack<int> SortStack(Stack<int> s)
{
    Stack<int> result = new Stack<int>();
    Stack<int> temp = new Stack<int>();
    while (!s.IsEmpty())
    {
        int x = s.Pop();
        temp.Push(x);              // to restore s at the end
        InsertSorted(result, x);
    }
    while (!temp.IsEmpty())
        s.Push(temp.Pop());
    return result;
}

ג. O(n²): n זימונים של InsertSorted, וכל אחד עד O(n).

ד. מחסנית שהגדול ביותר בראש, בסדר יורד: (top) 9, 7, 4, 1 (bottom). כל איבר שמוצא קטן מכל מה שכבר בתוצאה, ולכן InsertSorted דוחפת אותו מיד לראש, בלי להוציא אף איבר. סך הכול O(n).

public static Job MaxPriority(Stack<Job> jobs)
{
    Stack<Job> temp = new Stack<Job>();
    Job max = jobs.Top();
    while (!jobs.IsEmpty())
    {
        Job j = jobs.Pop();
        if (j.GetPriority() > max.GetPriority())   // > keeps the one closest to the top
            max = j;
        temp.Push(j);
    }
    while (!temp.IsEmpty())
        jobs.Push(temp.Pop());
    return max;
}

public static void RemoveBelow(Stack<Job> jobs, int p)
{
    Stack<Job> temp = new Stack<Job>();
    while (!jobs.IsEmpty())
    {
        Job j = jobs.Pop();
        if (j.GetPriority() >= p)
            temp.Push(j);
    }
    while (!temp.IsEmpty())
        jobs.Push(temp.Pop());
}

public static void PromoteMax(Stack<Job> jobs)
{
    Job max = MaxPriority(jobs);
    Stack<Job> temp = new Stack<Job>();
    while (!jobs.IsEmpty())
    {
        Job j = jobs.Pop();
        if (j != max)              // the same object, not the same priority
            temp.Push(j);
    }
    while (!temp.IsEmpty())
        jobs.Push(temp.Pop());
    jobs.Push(max);
}

ד. O(n): זימון אחד של MaxPriority (O(n)) ועוד מעבר אחד והחזרה (O(n)).

public static Stack<int> SegmentSums(Stack<int> s)
{
    Stack<int> temp = new Stack<int>();   // to restore s
    Stack<int> sums = new Stack<int>();   // reversed order
    int sum = 0;
    while (!s.IsEmpty())
    {
        int x = s.Pop();
        temp.Push(x);
        if (x == 0)
        {
            sums.Push(sum);
            sum = 0;
        }
        else
            sum += x;
    }
    sums.Push(sum);                        // the last segment
    while (!temp.IsEmpty())
        s.Push(temp.Pop());
    Stack<int> result = new Stack<int>();
    while (!sums.IsEmpty())
        result.Push(sums.Pop());
    return result;
}

public static int LongestSegment(Stack<int> s)
{
    int max = 0, len = 0;
    while (!s.IsEmpty())
    {
        if (s.Pop() == 0)
            len = 0;
        else
        {
            len++;
            if (len > max)
                max = len;
        }
    }
    return max;
}

ג. O(n): שלוש לולאות בזו אחר זו (מעבר, החזרה והיפוך), ואף אחת מהן לא עוברת יותר מ-n איברים.

public static Stack<int> Merge(Stack<int> s1, Stack<int> s2)
{
    Stack<int> temp = new Stack<int>();    // largest ends up on top
    while (!s1.IsEmpty() && !s2.IsEmpty())
    {
        if (s1.Top() <= s2.Top())
            temp.Push(s1.Pop());
        else
            temp.Push(s2.Pop());
    }
    while (!s1.IsEmpty())
        temp.Push(s1.Pop());
    while (!s2.IsEmpty())
        temp.Push(s2.Pop());
    Stack<int> result = new Stack<int>();
    while (!temp.IsEmpty())
        result.Push(temp.Pop());
    return result;
}

public static Stack<int> Common(Stack<int> s1, Stack<int> s2)
{
    Stack<int> temp = new Stack<int>();
    while (!s1.IsEmpty() && !s2.IsEmpty())
    {
        if (s1.Top() == s2.Top())
        {
            temp.Push(s1.Pop());
            s2.Pop();
        }
        else if (s1.Top() < s2.Top())
            s1.Pop();
        else
            s2.Pop();
    }
    Stack<int> result = new Stack<int>();
    while (!temp.IsEmpty())
        result.Push(temp.Pop());
    return result;
}

ג. O(n + m): כל איבר יוצא פעם אחת מהמחסנית שלו, ועובר פעם אחת לעזר ופעם אחת לתוצאה.

ד. O(n·m): לכל אחד מ-n האיברים יש מעבר O(m) על s2.
